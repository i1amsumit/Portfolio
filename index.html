Ola_Ride_Analytics_Queries.sql
// 1. Retrieve booking status counts 
CREATE VIEW Ride_Status_Summary AS
SELECT 
    Booking_Status,
    COUNT(Booking_ID) AS Total_Rides,
    ROUND(COUNT(Booking_ID) * 100.0 / (SELECT COUNT(*) FROM Ride_Bookings), 2) AS Percentage
FROM Ride_Bookings
GROUP BY Booking_Status;
// 2. Calculate Revenue & Avg Distance by Vehicle Type 
CREATE VIEW Vehicle_Performance AS
SELECT 
    Vehicle_Type,
    SUM(Booking_Value) AS Total_Revenue,
    AVG(Ride_Distance) AS Avg_Distance_KM,
    AVG(Customer_Rating) AS Avg_Rating
FROM Ride_Bookings
WHERE Booking_Status = 'Success'
GROUP BY Vehicle_Type;
// 3. Customer Cancellation Breakdown 
SELECT 
    Canceled_Rides_by_Customer,
    COUNT(*) AS Cancellation_Count
FROM Ride_Bookings
WHERE Canceled_Rides_by_Customer IS NOT NULL
GROUP BY Canceled_Rides_by_Customer
ORDER BY Cancellation_Count DESC;
